CREATE OR ALTER PROCEDURE SP_ValidarLoginAdmin
(
    @Usuario VARCHAR(50),
    @Contrasena VARCHAR(255)
)
AS
BEGIN
    SET NOCOUNT ON;

    SELECT TOP 1
        IdUsuario,
        IdEmpresa,
        Usuario,
        PasswordHash,
        NombreCompleto,
        EsSuperAdmin,
        Estado,
        FechaRegistro
    FROM AdminUsuario_GT
    WHERE Usuario = @Usuario
      AND PasswordHash = @Contrasena
      AND Estado = 1;
END
GO

alter PROCEDURE SP_SincronizarEmpresas
AS
BEGIN
    SET NOCOUNT ON;
    --generar tabla virtual
    CREATE TABLE #Empresas
    (
        BaseDatos VARCHAR(128),
        ID_EMPRESA INT,
        COD_EMPRESA VARCHAR(50),
        RUC_EMPRESA VARCHAR(20),
        RAZONSOCIAL_EMPRESA VARCHAR(250),
    
        AlmacenesMax INT,
        UbicacionesMax INT,
        ProductosMax INT,
        UsuariosAdminMax INT,
        UsuariosOperadorMax INT,
        InventariosPreparadosMax INT
    );

    -- inicializar variables de entrada
    DECLARE @BD VARCHAR(128);
    DECLARE @SQL NVARCHAR(MAX);

    --hacer un cursor para recorrer todas las bd que enpiezen por INVENTARIO_
    DECLARE cBD CURSOR FOR
    SELECT name
    FROM sys.databases
    WHERE name LIKE 'INVENTARIO_%'
      AND state = 0;

      --abrimos cursor
    OPEN cBD;

    FETCH NEXT FROM cBD INTO @BD;

    WHILE @@FETCH_STATUS = 0
    BEGIN
    -- consulta interna en variable
        SET @SQL = '
        IF EXISTS (
            SELECT 1
            FROM [' + @BD + '].sys.tables
            WHERE name = ''ASF_EMPRESA''
        )
        BEGIN

            INSERT INTO #Empresas
        (
        BaseDatos,
        ID_EMPRESA,
        COD_EMPRESA,
        RUC_EMPRESA,
        RAZONSOCIAL_EMPRESA,
        AlmacenesMax,
        UbicacionesMax,
        ProductosMax,
        UsuariosAdminMax,
        UsuariosOperadorMax,
        InventariosPreparadosMax
        )
            SELECT
                ''' + @BD + ''',
                E.ID_EMPRESA,
                E.COD_EMPRESA,
                E.RUC_EMPRESA,
                E.RAZONSOCIAL_EMPRESA,
                L.AlmacenesMax,
                L.UbicacionesMax,
                L.ProductosMax,
                L.UsuariosAdmiMax,
                L.UsuariosOpeMax,
                L.InventariosPreparadosMax
                        FROM [' + @BD + '].dbo.ASF_EMPRESA E
                        LEFT JOIN [' + @BD + '].dbo.ASF_CONFIG_LICENCIA L
                    ON E.ID_EMPRESA = L.IdEmpresa
        END';

        BEGIN TRY
        PRINT @SQL; -- ver 
            EXEC(@SQL);
        END TRY
        BEGIN CATCH
            PRINT 'Error leyendo BD: ' + @BD;
        END CATCH

        FETCH NEXT FROM cBD INTO @BD;
    END

    CLOSE cBD;
    DEALLOCATE cBD;

    SELECT
    BaseDatos,
    COD_EMPRESA,
    AlmacenesMax,
    UbicacionesMax,
    ProductosMax
FROM #Empresas

SELECT *
FROM #Empresas

    -----------------------------------------------------
    -- INSERTAR NUEVAS EMPRESAS
    -----------------------------------------------------

    INSERT INTO Empresa_GT
(
   
    CodEmpresa,
    Ruc,
    Nombre,
    Estado,
    FechaRegistro
)
SELECT
    
    E.COD_EMPRESA,
    E.RUC_EMPRESA,
    E.RAZONSOCIAL_EMPRESA,
    1,
    GETDATE()
FROM #Empresas E
WHERE NOT EXISTS
(
    SELECT 1
    FROM Empresa_GT G
    WHERE G.Ruc = E.RUC_EMPRESA
);

    -----------------------------------------------------
    -- REGISTRAR INSTANCIAS
    -----------------------------------------------------

    INSERT INTO InstanciaCliente_GT
    (
        IdEmpresa,
        NombreBD,
        ServidorSql,
        RutaWeb,
        Estado
    )
    SELECT
        G.IdEmpresa,
        E.BaseDatos,
        @@SERVERNAME,
        '',
        1
    FROM #Empresas E
    INNER JOIN Empresa_GT G
        ON G.Ruc = E.RUC_EMPRESA
    WHERE NOT EXISTS
    (
        SELECT 1
        FROM InstanciaCliente_GT I
        WHERE I.NombreBD = E.BaseDatos
    );

    -----------------------------------------------------
    -- REGISTRAR EN LA BD LA LICENCIA POR PARTE DE CONFIG
    -----------------------------------------------------
    INSERT INTO ConfigLicencia_GT
        (
            IdEmpresa,
            AlmacenesMax,
            UbicacionesMax,
            ProductosMax,
            UsuariosAdminMax,
            UsuariosOperadorMax,
            InventariosPreparadosMax
        )
    SELECT
            G.IdEmpresa,
            E.AlmacenesMax,
            E.UbicacionesMax,
            E.ProductosMax,
            E.UsuariosAdminMax,
            E.UsuariosOperadorMax,
            E.InventariosPreparadosMax
        FROM #Empresas E
            INNER JOIN Empresa_GT G
        ON G.Ruc = E.RUC_EMPRESA
            WHERE NOT EXISTS
                (
                    SELECT 1
                    FROM ConfigLicencia_GT C
                    WHERE C.IdEmpresa = G.IdEmpresa
                );


    ------------------------------------------------------
    -- Actualizar licencias por medio de joins
    ------------------------------------------------------

    UPDATE C
    SET
        C.AlmacenesMax = E.AlmacenesMax,
        C.UbicacionesMax = E.UbicacionesMax,
        C.ProductosMax = E.ProductosMax,
        C.UsuariosAdminMax = E.UsuariosAdminMax,
        C.UsuariosOperadorMax = E.UsuariosOperadorMax,
        C.InventariosPreparadosMax = E.InventariosPreparadosMax
            FROM ConfigLicencia_GT C
                INNER JOIN Empresa_GT G
            ON C.IdEmpresa = G.IdEmpresa
                INNER JOIN #Empresas E
            ON G.Ruc = E.RUC_EMPRESA;
    -----------------------------------------------------
    -- RESULTADO
    -----------------------------------------------------

    SELECT
        G.IdEmpresa,
        G.CodEmpresa,
        G.Ruc,
        G.Nombre,
        I.NombreBD
    FROM Empresa_GT G
    LEFT JOIN InstanciaCliente_GT I
        ON G.IdEmpresa = I.IdEmpresa
    ORDER BY G.Nombre;

    --borrar tabla temporal
    IF OBJECT_ID('tempdb..#Empresas') IS NOT NULL
    DROP TABLE #Empresas;
END
GO


EXEC SP_SincronizarEmpresas

CREATE OR ALTER PROCEDURE SP_ActualizarLicenciaCliente
(
    @IdEmpresa BIGINT
)
AS
BEGIN

    SET NOCOUNT ON;

    DECLARE @NombreBD VARCHAR(100);
--     DECLARE @IdEmpresaOrigen INT;

    DECLARE @AlmacenesMax INT;
    DECLARE @UbicacionesMax INT;
    DECLARE @ProductosMax INT;
    DECLARE @UsuariosAdminMax INT;
    DECLARE @UsuariosOperadorMax INT;
    DECLARE @InventariosPreparadosMax INT;

    DECLARE @SQL NVARCHAR(MAX);

    --------------------------------------------------
    -- VALIDAR EMPRESA
    --------------------------------------------------

    IF NOT EXISTS
    (
        SELECT 1
        FROM Empresa_GT
        WHERE IdEmpresa = @IdEmpresa
    )
    BEGIN
        RAISERROR('Empresa no encontrada.',16,1);
        RETURN;
    END

    --------------------------------------------------
    -- OBTENER BD CLIENTE
    --------------------------------------------------

    SELECT
    @NombreBD = I.NombreBD
FROM InstanciaCliente_GT I
WHERE I.IdEmpresa = @IdEmpresa;

    IF @NombreBD IS NULL
    BEGIN
        RAISERROR('No existe instancia registrada.',16,1);
        RETURN;
    END

    --------------------------------------------------
    -- OBTENER LICENCIA MAESTRA
    --------------------------------------------------

    SELECT
        @AlmacenesMax = AlmacenesMax,
        @UbicacionesMax = UbicacionesMax,
        @ProductosMax = ProductosMax,
        @UsuariosAdminMax = UsuariosAdminMax,
        @UsuariosOperadorMax = UsuariosOperadorMax,
        @InventariosPreparadosMax = InventariosPreparadosMax
    FROM ConfigLicencia_GT
    WHERE IdEmpresa = @IdEmpresa;

    --------------------------------------------------
    -- ACTUALIZAR CLIENTE
    --------------------------------------------------

    SET @SQL = '
UPDATE [' + @NombreBD + '].dbo.ASF_CONFIG_LICENCIA
SET
    AlmacenesMax = ' + CAST(@AlmacenesMax AS VARCHAR(20)) + ',
    UbicacionesMax = ' + CAST(@UbicacionesMax AS VARCHAR(20)) + ',
    ProductosMax = ' + CAST(@ProductosMax AS VARCHAR(20)) + ',
    UsuariosAdmiMax = ' + CAST(@UsuariosAdminMax AS VARCHAR(20)) + ',
    UsuariosOpeMax = ' + CAST(@UsuariosOperadorMax AS VARCHAR(20)) + ',
    InventariosPreparadosMax = ' + CAST(@InventariosPreparadosMax AS VARCHAR(20)) + ';
';
    PRINT @SQL; -- ver
    EXEC(@SQL);

END
GO

EXEC SP_ActualizarLicenciaCliente 1;
GO









CREATE OR ALTER TRIGGER TR_ConfigLicencia_GT_Update
ON ConfigLicencia_GT
AFTER UPDATE
AS
BEGIN

    SET NOCOUNT ON;

    DECLARE @IdEmpresa BIGINT;

    DECLARE c CURSOR FOR
    SELECT DISTINCT IdEmpresa
    FROM inserted;

    OPEN c;

    FETCH NEXT FROM c INTO @IdEmpresa;

    WHILE @@FETCH_STATUS = 0
    BEGIN

        EXEC SP_ActualizarLicenciaCliente @IdEmpresa;

        FETCH NEXT FROM c INTO @IdEmpresa;

    END

    CLOSE c;
    DEALLOCATE c;

END
GO
